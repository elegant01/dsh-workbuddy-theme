// 拉取会话列表并按时间排序
async function loadSessions(profile, limit = 20) {
  const res = await fetch(`/api/profiles/${profile}/sessions`);
  if (!res.ok) throw new Error('加载失败');
  const list = await res.json();
  return list
    .sort((a, b) => b.mtime - a.mtime)
    .slice(0, limit);
}
